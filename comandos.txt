--count              : QUANTIDADE TOTAL DE CONTAS A SEREM CRIADAS
--parallel           : QUANTIDADE DE BOTS/JANELAS EM PARALELO
--pix-type           : TIPO DE CHAVE DE SAQUE (cpf, telefone, email)
--target             : VALOR ALVO DA BANCA (PARA PARAR DE JOGAR)
--platform           : NOME DA PLATAFORMA (EX: 448bet, hwbet)
--invite-id          : ID DO CONVITE (AFILIADO)
--sms 1              : ATIVA VERIFICAÇÃO SMS (PRECISA DE SALDO NA SMS24H)
--gee-token-individual : CADA CONTA GERA SEU PRÓPRIO TOKEN DE SEGURANÇA (RECOMENDADO)
--no-save-db         : NÃO SALVA AS CONTAS NO BANCO DE DADOS LOCAL (SQLITE)
--no-proxy           : DESATIVA O USO DE PROXIES (RODA NO SEU IP LOCAL)
--gem-trava          : TENTA "TRAVAR" O BÔNUS NO JOGO GEM SAVIOUR (SALDO 15-1000)
--hood-trava         : TENTA "TRAVAR" O BÔNUS NO JOGO HOOD VS WOLF (SALDO 15-1000)
--double             : ATIVA TRAVA DOUBLE FORTUNE (EX: --double "fortune-trava-bet 0.9")
--ganesha            : ATIVA TRAVA GANESHA GOLD (EX: --ganesha "ganesha-trava-bet 2.0")
--emperor            : ATIVA TRAVA EMPEROR'S FAVOUR (EX: --emperor "emperor-trava-bet 14.40")
--sunshine           : ATIVA TRAVA SUNSHINE COAST (EX: --sunshine "sunshine-armar-bet 15" OU --sunshine "sunshine-aleatoria 15")
--journey            : ATIVA TRAVA JOURNEY TO THE WEALTH (EX: --journey "journey-trava-bet 12")
--firestrike         : ATIVA MODO SUBIR SALDO FIRE STRIKE 2 ALL-WIN (PURA ALAVANCAGEM)
--fire88             : ATIVA MODO SUBIR SALDO FIRE 88 ALL-WIN (PURA ALAVANCAGEM, 7 LINHAS)
--spirit             : ATIVA TRAVA SPIRIT OF ADVENTURE (EX: --spirit "spirit-trava-bet 10")
--voodoo             : ATIVA TRAVA VOODOO MAGIC (EX: --voodoo "voodoo-trava-bet 0.40")
--muertos            : ATIVA TRAVA MUERTOS MULTIPLIER MEGAWAYS (EX: --muertos "muertos-trava-bet 0.40")
--diamond            : ATIVA TRAVA DIAMOND CLUSTER LUCK (EX: --diamond "diamond-trava-bet 0.40")
--charm              : ATIVA TRAVA LUCKY CHARM (EX: --charm "charm-trava-bet 0.40")
--treasure-house     : ATIVA TRAVA TREASURE HOUSE (EX: --treasure-house "treasure-house-trava-bet 2.0")
--codigo             : RESGATA O CÓDIGO DO TELEGRAM EM TODAS AS CONTAS (SUBSTITUI BÔNUS PADRÃO)
--aguardar-codigo    : MODO 3 FASES INTERATIVO: (1) CRIA CONTAS+SAQUE → (2) PARA NO TERMINAL E PEDE O CÓDIGO → (3) RESGATA EM PARALELO → PERGUNTA PARALELO DE JOGO → JOGA SÓ NAS CONTAS COM SALDO
--bet-aleatoria      : ESCOLHE UMA BET ALEATÓRIA ACIMA DE 12 NOS JOGOS SUPORTADOS
--10-cetavo          : ATIVA RECUPERAÇÃO NO TRIPLE TIGERS COM SALDO < 0.30 (APOSTA DE 0.10)
